secret_number = 42
attempts = 0

while attempts < 5:
    guess = int(input("Guess a number between 1 and 100: "))

    if guess < 1 or guess > 100:
        continue

    attempts += 1

    if guess == secret_number:
        print("Congratulations! You guessed correctly.")
        break
    else:
    print("Wrong guess! Try again.")

else:
     print("Game over! you lost.")
